name: CI

on:
  push:
    branches: [main]
  pull_request:

jobs:
  build:
    runs-on: ubuntu-latest
    strategy:
      matrix:
        dir: [services/backend, services/frontend]
    steps:
      - uses: actions/checkout@v4
      - name: Setup Node
        uses: actions/setup-node@v4
        with:
          node-version: 20
      - name: Install and Test
        working-directory: ${{ matrix.dir }}
        run: |
          npm ci
          npm run build --if-present
          npm test --if-present

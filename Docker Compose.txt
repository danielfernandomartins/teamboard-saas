version: '3.8'
services:
  db:
    image: postgres:15
    environment:
      POSTGRES_USER: postgres
      POSTGRES_PASSWORD: postgres
      POSTGRES_DB: teamboard
    ports:
      - "5432:5432"
    volumes:
      - db_data:/var/lib/postgresql/data
  backend:
    build: ./services/backend
    environment:
      DATABASE_URL: postgres://postgres:postgres@db:5432/teamboard
    ports:
      - "4000:4000"
    depends_on:
      - db
  frontend:
    build: ./services/frontend
    ports:
      - "3000:3000"
    depends_on:
      - backend
